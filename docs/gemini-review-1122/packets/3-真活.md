/** 出题人是用自然语言说要什么样的人（「我要一个能看懂人工智能行业的人来做投资」），还是给了个职位名 */

/** 限时：15–240 分钟（以前出题卡 120、修订也卡 120，出任务的人说改 180 改不动，2026-11-03；11-13 下限放到 15：课堂、工作坊发出去的小体验） */
export const clampMinutes = (v: any, d: number) => Math.min(240, Math.max(15, Math.round(Number(v) || d)));
/** 没说时长就 60 分钟（11-13 潜历：模型自己在 45–90 里挑，默认太久） */
export const DEFAULT_MINUTES = 60;
/** 改限时：计时器用的 time_minutes 和任务书 / 规则里写的「120 分钟」「2 小时」一起换，别出现计时器和题面对不上 */
export function withMinutes(task: any, v: any) {
  const old = Number(task?.time_minutes) || 60, mins = clampMinutes(v, old);
  const label = mins % 60 === 0 ? `${mins / 60} 小时` : `${mins} 分钟`;
  const swap = (x: string) => { let y = String(x || '').replace(new RegExp(`${old}\\s*分钟`, 'g'), `${mins} 分钟`).replace(new RegExp(`\\b${old}(\\s*|-)(minutes?|mins?)\\b`, 'gi'), (_m, sep, u) => `${mins}${sep}${u}`); if (old % 60 === 0) y = y.replace(new RegExp(`${old / 60}\\s*个?小时`, 'g'), label); return y; };
  return { old, mins, task: { ...task, time_minutes: mins, brief: swap(task?.brief), rules: (task?.rules || []).map(swap) } };
}

/** 原题直出 / 参照原题后清掉原企业痕迹（11-05） */
export function scrubOrigin(task: WorkTask, origin: string[], name: string) {
  const esc = (x: string) => x.replace(/[.*+?^${}()|[\]\\]/g, '\\$&');
  const re = new RegExp(origin.map(esc).join('|'), 'g');
  const has = new RegExp(re.source);   // 判断用不带 g 的（带 g 的 test 会记位置，逐句判断时漏判）
  const basis = (b?: string) => { if (!b) return b; const keep = b.split(/(?<=[；;。\n])/).filter(x => !has.test(x) && !/原题|参照|对标|照搬|任务库/.test(x)); const out = keep.join('').trim(); return out || '按任务库同类任务的结构，结合本企业公开信息生成'; };
  const swap = (v: any): any => typeof v === 'string' ? v.replace(re, name) : Array.isArray(v) ? v.map(swap) : v && typeof v === 'object' ? Object.fromEntries(Object.entries(v).map(([k, x]) => [k, swap(x)])) : v;
  for (const m of task.materials) m.basis = basis(m.basis);
  for (const r of task.rubric) r.basis = basis(r.basis) || '';
  task.problem = swap(task.problem); task.task = swap(task.task); task.materials = swap(task.materials); task.rubric = swap(task.rubric);
}

/** 改岗位名 / 要找的人：work_tasks.role 和 task.role 都换；题面、问题里出现的旧名字一起换（不经模型改写） */
export function withRole(t: { role?: string; task?: any; problem?: any }, v: any) {
  const role = String(v || '').trim().slice(0, 120); if (!role) throw new Error('岗位名不能为空');
  const olds = [...new Set([t.role, t.task?.role].map(x => String(x || '').trim()).filter(x => x && x !== role))].sort((a, b) => b.length - a.length);
  const swap = (x: any) => { let s = JSON.stringify(x ?? null); for (const o of olds) s = s.split(JSON.stringify(o).slice(1, -1)).join(JSON.stringify(role).slice(1, -1)); return JSON.parse(s); };
  return { old: t.role || '', role, task: { ...swap(t.task), role }, problem: swap(t.problem) };
}

/** 改企业名（11-18：企业库名称写错、数据部更正后，已出的题要跟着改）：页头 company_name、出题依据里的企业画像快照、题面 / 问题 / 材料 / 评分标准里的旧名一起换 */
export function withCompany(t: { company_name?: string; task?: any; problem?: any; materials?: any; rubric?: any; context?: any }, v: any) {
  const name = String(v || '').trim().slice(0, 120); if (!name) throw new Error('企业名不能为空');
  const olds = [...new Set([t.company_name, t.context?.company?.name].map(x => String(x || '').trim()).filter(x => x.length >= 2 && x !== name && !name.includes(x)))].sort((a, b) => b.length - a.length);
  const swap = (x: any) => { if (x == null) return x; let s = JSON.stringify(x); for (const o of olds) s = s.split(JSON.stringify(o).slice(1, -1)).join(JSON.stringify(name).slice(1, -1)); return JSON.parse(s); };
  const context = t.context ? { ...swap(t.context), ...(t.context.company ? { company: { ...swap(t.context.company), name } } : {}) } : t.context;
  return { old: t.company_name || '', olds, company_name: name, task: swap(t.task), problem: swap(t.problem), materials: swap(t.materials), rubric: swap(t.rubric), context };
}

/** 带兜底链的 JSON 调用：Opus → Sonnet → 千问；记账 tool=work */
/** onPart（11-15）：边生成边把写完的顶层字段交出来（出题分段回传）；换模型重试时会从头再交一遍 */
export async function askJson(prompt: string, model: string, task_name: string, institution?: string, extra?: Record<string, any>, onPart?: (key: string, value: any) => void): Promise<any> {
  const chain = [model, ...DOMESTIC_FALLBACK.filter(m => m !== model)];
  let lastErr: any = null;
  for (const m of chain) {
    try {
      const res = await generateContent(prompt, m, { jsonMode: true, ...(onPart ? { onText: jsonParts(onPart) } : {}) });
      await logTokenUsage({ tool_name: 'work', task_name, institution: institution || '', model_id: m, usageMetadata: res.usageMetadata, ...(extra || {}) } as any).catch(() => {});
      const p = parseJsonLoose(res.text);
      if (!p || typeof p !== 'object') throw new Error('返回不是 JSON 对象');
      return { parsed: p, model: m };
    } catch (e) { lastErr = e; console.warn(`[work] ${m} 失败，换下一个：${String((e as any)?.message).slice(0, 120)}`); }
  }
  throw lastErr || new Error('生成失败');
}

/** 第一步：把企业 / 岗位 / 职业 / 标准四层材料拼起来（都是库里的，带来源） */
export async function gatherContext(companyQuery: string, role: string, opts: { intro?: string } = {}): Promise<WorkContext> {

    // 贴了介绍 = 出题人说这家不在库里：只认名字完全对上的，不拿模糊匹配的别家顶替
    company = (data || []).find((c: any) => c.name === q || c.brief_name === q) || (intro ? null : data?.[0]) || null;
  }
  const J = 'name, title_cn, department, job_function, summary_cn, responsibilities, overview, skill_labels, education_description, major_requirement, preferred_qualifications, city';
  if (company) {
    ctx.company = company;
    const cid = company.id;
    const [{ data: products }, { data: news }, { data: execs }, { data: jobs }] = await Promise.all([
      db.from('company_products').select('name, category, description, is_flagship').eq('company_id', cid).eq('if_delete', false).order('is_flagship', { ascending: false }).limit(8),
      db.from('company_news').select('description, publish_date_str, kind').eq('company_id', cid).eq('if_delete', false).order('publish_date', { ascending: false, nullsFirst: false }).limit(8),
      db.from('company_executives').select('name, title').eq('company_id', cid).eq('if_delete', false).limit(5),
      db.from('jobs').select(J).eq('company_id', cid).eq('if_delete', false).or(`name.ilike.%${role}%,title_cn.ilike.%${role}%,job_function.ilike.%${role}%`).limit(3),

    ctx.products = products || []; ctx.news = news || []; ctx.executives = execs || []; ctx.jobs = jobs || [];

      const { data } = await db.from('jobs').select(J).eq('company_id', cid).eq('if_delete', false).not('responsibilities', 'is', null).limit(80);
      const grams = new Set<string>(); const rr = role.replace(/[\s，。、“”「」（）()的人来做一个能]/g, ''); for (let i = 0; i + 2 <= rr.length; i++) grams.add(rr.slice(i, i + 2));
      const score = (j: any) => { const t = `${j.name || ''} ${j.title_cn || ''} ${j.department || ''} ${j.job_function || ''} ${String(j.responsibilities || '').slice(0, 600)}`; let n = 0; for (const g of grams) if (t.includes(g)) n++; if (/投资|投后|研究|分析|行业/.test(t) && /投资|研究|分析|行业/.test(rr)) n += 2; if (/Portfolio|被投企业/.test(t)) n -= 2; return n; };

    }
    // 深度画像：每个专题压成一段（出题时最有用的材料：投资机构的被投企业、合伙人、招聘要求……）

  return ctx;
}

/** 关联主体：和本企业共同高管 ≥2 位的企业（同一体系的品牌 / 主体），最多 2 家（2026-10-22：武岳峰资本 ↔ 鼎峰科创） */
async function relatedEntities(cid: number, label: Record<string, string>, strip: (v: any) => any): Promise<NonNullable<WorkContext['related']>> {
  const { data: ex } = await db.from('company_executives').select('name').eq('company_id', cid).eq('if_delete', false);
  const names = [...new Set((ex || []).map(e => e.name).filter(Boolean))];
  if (!names.length) return [];
  const { data: other } = await db.from('company_executives').select('company_id, name').in('name', names).neq('company_id', cid).eq('if_delete', false);
  const cnt = new Map<number, Set<string>>();
  for (const o of other || []) { if (!cnt.has(o.company_id)) cnt.set(o.company_id, new Set()); cnt.get(o.company_id)!.add(o.name); }
  const out: NonNullable<WorkContext['related']> = [];
  for (const [rid, shared] of [...cnt.entries()].filter(([, x]) => x.size >= 2).sort((a, b) => b[1].size - a[1].size).slice(0, 2)) {
    const { data: r } = await db.from('companies').select('id, name, aliases, one_sentence, introduction, industry_position').eq('id', rid).eq('if_delete', false).maybeSingle();
    if (!r) continue;
    const [{ data: news }, { data: deep }] = await Promise.all([
      db.from('company_news').select('description, publish_date_str').eq('company_id', rid).eq('if_delete', false).order('publish_date', { ascending: false, nullsFirst: false }).limit(8),
      db.from('company_deep_research').select('topic, data').eq('company_id', rid),
    ]);
    out.push({ id: r.id, name: r.name, shared: [...shared],
      text: [r.aliases?.length ? `别名：${r.aliases.join('、')}` : '', r.one_sentence, r.introduction, r.industry_position].filter(Boolean).join('\n').slice(0, 1500),
      news: (news || []).map(n => `${n.publish_date_str || ''} ${s(n.description, 240)}`),
      deep: (deep || []).filter(d => d.data && JSON.stringify(d.data).length > 30).map(d => `${label[d.topic] || d.topic}：${JSON.stringify(strip(d.data)).slice(0, 900)}`) });
  }
  return out;
}

/** 出题人贴的企业介绍 → 和企业库同样字段的画像（只用介绍里有的，不编） */
export async function parseCompanyIntro(name: string, intro: string): Promise<{ company: Record<string, any>; products: WorkContext['products']; news: WorkContext['news']; executives: WorkContext['executives'] }> {
  const prompt = `下面是出题人贴的一段企业介绍（可能来自官网、BP、公众号或口述）。把它整理成企业画像 JSON，给出题用。
规则：只用介绍里写到的信息，没写的字段留空字符串 / 空数组，绝不编造数字、客户、人名；原话里的关键事实尽量保留。
企业名：${name}
## 介绍原文
${intro.slice(0, 6000)}
## 输出 JSON
{"company":{"name":"","industry":"行业","city":"","company_scale":"规模 / 人数 / 阶段（如 A 轮、初创 20 人）","one_sentence":"一句话说清它做什么","introduction":"简介","business_profile":"业务与客户","product_area":"产品领域","tech_advantage":"技术或能力优势","growth_signals":"正在推进的事：新业务、出海、融资、扩张、招人原因","industry_position":"在行业里的位置、竞争对手","public_sentiment":"介绍里提到的难点、风险或卡点"},
"products":[{"name":"","description":""}],"news":[{"publish_date_str":"","description":"近期动作"}],"executives":[{"name":"","title":""}]}`;
  const { parsed: p } = await askJson(prompt, 'qwen-plus', 'Generate · 解析企业介绍', name);
  const arr = (v: any) => Array.isArray(v) ? v : [];
  return {
    company: Object.fromEntries(Object.entries(p.company || {}).filter(([, v]) => typeof v === 'string' && v.trim())),
    products: arr(p.products).filter((x: any) => x?.name).slice(0, 8),
    news: arr(p.news).filter((x: any) => x?.description).slice(0, 8),

/** 把上传的附件挂成材料（kind=file）：有文字的带文字（评卷和 AI 做题能读），图片只有链接和说明 */
/**
 * 附件去重（11-05 潜历：「聊着出任务上传的附件，做题的人看不到」）：出题 AI 常另写一份文字材料叫「附件1：潜仔和小鸥.png（原始图片）」，
 * 真文件排在最后——候选人点开「附件1」只看到文字。合成一份：留 AI 写的说明和它的编号，挂上原文件；题面里指向真文件编号的改指过来。
 */
export function mergeAttachmentDupes<T extends { problem?: any; task?: any; rubric?: any; materials: any[] }>(t: T): T {
  const mats = [...(t.materials || [])]; let changed = false; const remap: Record<string, string> = {};
  for (const f of mats.filter(m => m.kind === 'file' && m.url)) {
    const nm = String(f.name || f.title || ''); const stem = nm.replace(/\.[A-Za-z0-9]{1,6}$/, '');
    if (stem.length < 2) continue;
    const d = mats.find(m => m !== f && m.kind !== 'file' && !m.url && (String(m.title || '').includes(nm) || String(m.title || '').includes(stem)));
    if (!d) continue;
    Object.assign(d, { kind: 'file', url: f.url, mime: f.mime, name: f.name || nm, content: [d.content, f.content].filter(Boolean).join('\n\n').slice(0, 20000), source: 'upload' });
    mats.splice(mats.indexOf(f), 1); remap[f.id] = d.id; changed = true;
  }
  if (!changed) return t;
  const re = new RegExp(`\\b(${Object.keys(remap).join('|')})\\b`, 'g');
  const swap = (v: any): any => typeof v === 'string' ? v.replace(re, x => remap[x] || x) : Array.isArray(v) ? v.map(swap) : v && typeof v === 'object' ? Object.fromEntries(Object.entries(v).map(([k, x]) => [k, swap(x)])) : v;
  return { ...t, materials: mats, ...(t.task ? { task: swap(t.task) } : {}), ...(t.problem ? { problem: swap(t.problem) } : {}), ...(t.rubric ? { rubric: swap(t.rubric) } : {}) };
}

    if (want.length) { task.task.deliverables = task.task.deliverables.slice(0, want.length).map((d, i) => ({ ...d, module: want[i] as any })); if (!task.task.deliverables.length) task.task.deliverables = want.map((m, i) => ({ id: `d${i + 1}`, title: '按任务要求交付', format: m, module: m as any })); }
  }
  // 照任务库原题出的：原题企业名不许留下（客户在任务空间能看到「取材依据」）——取材依据里提到原题 / 原企业的句子删掉，正文里的原企业名换成新企业
  if (opts.base?.origin?.length) scrubOrigin(task, opts.base.origin, name);
  // 权重归一到 100；依赖只认存在的条目
  const ids = new Set(task.rubric.map(r => r.id));
  const sum = task.rubric.reduce((a, r) => a + r.weight, 0) || 1;
  for (const r of task.rubric) { r.weight = Math.round(r.weight / sum * 100); r.depends_on = r.depends_on.filter(d => ids.has(d) && d !== r.id); }
  // 出题的内部约束（企业负面舆情不进题等）是给出题模型看的，不许印进候选人看的规则——那句话本身就点出了负面话题（11-05 国内版验收）
  task.task.rules = task.task.rules.filter(r => !/负面|舆情/.test(r) && !(/(严禁|不得|不要|禁止|切勿).{0,12}(提及|涉及|出现|谈及)/.test(r) && /监管|减持|丑闻|传闻|亏损|离职|处罚|立案/.test(r)));
  if (!task.task.rules.some(r => /AI/.test(r))) task.task.rules.unshift(en ? 'AI use: You may use AI freely — it is not cheating; we assess how you use it.' : 'AI 使用：可以随便用 AI，不算作弊；评的是你怎么用、什么是你自己判断的');
  task.task.style = { deep, steps: staged, hints: rich ? 'rich' : 'lean' };
  if (en) task.task.lang = 'en';
  // 案例推进式：步骤归一，测算在代码里真算一遍（模型给的「答案」和「数据」对不上是常事）
  if (staged) {
    const { stages, fixed, failed } = verifyCalcs(normalizeStages(p.stages));
    if (stages.length) task.task.stages = stages;
    if (fixed || failed) console.warn(`[work] 测算核对：改了 ${fixed} 个值，${failed} 个算式算不出`);
  }
  return { task, model: used };
}

    if (weak.length && me.score > 59) { me.score = 59; me.level = 'L2'; me.blocked_by = weak.join('、'); }
  }
  const total = Math.round(items.reduce((a, i) => a + i.score * i.weight, 0) / (items.reduce((a, i) => a + i.weight, 0) || 1));
  return { total, items, summary: sl(task)(p.summary, 500), strengths: (Array.isArray(p.strengths) ? p.strengths : []).slice(0, 4).map((x: any) => sl(task)(x, 120)), gaps: (Array.isArray(p.gaps) ? p.gaps : []).slice(0, 4).map((x: any) => sl(task)(x, 120)), who: opts.who };
}

/** 交付物里原样来自 AI 回答的比例：交付物按句切，长度 ≥ 14 的句子在任一 AI 回答里原样出现就算 */
function verbatimRatio(deliverable: string, answers: string[]) {
  const sents = deliverable.split(/[。！？\n；;]+/).map(x => x.replace(/\s+/g, '').trim()).filter(x => x.length >= 14);
  if (!sents.length) return 0;
  const pool = answers.map(a => a.replace(/\s+/g, ''));
  const hit = sents.filter(x => pool.some(a => a.includes(x))).reduce((n, x) => n + x.length, 0);
  return Math.round((hit / sents.reduce((n, x) => n + x.length, 0)) * 100);
}
/** 这道任务的 AI 使用标准：存在 work_tasks.task.ai_policy；没有就按依据现生成、存回去（t 是 work_tasks 的整行） */
export async function ensureAiPolicy(t: any): Promise<AiPolicy | null> {
  if (t?.task?.ai_policy?.v) return t.task.ai_policy as AiPolicy;
  try {
    const task = { env: t.task?.env || { type: 'doc' }, problem: t.problem, task: t.task, materials: t.materials || [], rubric: t.rubric || [] } as any;
    const policy = await buildAiPolicy(task, t.context || {}, askJson);
    const { data: now } = await db.from('work_tasks').select('task').eq('id', t.id).single();
    await db.from('work_tasks').update({ task: { ...(now?.task || t.task), ai_policy: policy } }).eq('id', t.id);
    return policy;
  } catch (e) { console.warn('[work-grade] AI 使用标准没生成出来，按通用口径评', String((e as any)?.message).slice(0, 120)); return null; }
}

/** 交卷时选填的「如果再给一天，先改哪里」：存成事件 next_step（2026-11-02，借潜历「迭代与判断」的一问） */
export function nextStepOf(events: WorkEvent[]): string {
  const e = [...events].reverse().find(x => x.type === 'next_step' && (x as any).text);
  return e ? String((e as any).text).slice(0, 1000) : '';
}

export async function buildReport(task: WorkTask, baseline: { result?: ResultReport } | null, deliverable: string, events: WorkEvent[], aiChat: { t: number; q: string; a: string }[], institution?: string, audit?: any[], policy?: AiPolicy | null, sid?: string) {

  const saved = (task.task as any).fluency_profile || null, id = (task as any).id as string | undefined;

  }
  return rep;
}
