export const isDepth = (v: any): v is Depth => v === 'light' || v === 'standard' || v === 'deep';
/**
 * 每段多深：人选了就用人选的；没选就按这一天几格自己判断（11-19 用户：要会判断长任务还是短任务）——
 * 段少每段就该长：≤ 3 格写长段（起承转合），4–5 格写标准段，6 格以上按形态默认。count = 这一天的格数（排骨架时是要排的格数，写章节时是已有章节数）
 */
export const depthOf = (jdSnapshot: any, count = 0): Depth | null => {
  if (isDepth(jdSnapshot?.depth)) return jdSnapshot.depth;
  const n = slotsOf(jdSnapshot) || count;
  return n >= 2 && n <= 3 ? 'deep' : n >= 4 && n <= 5 ? 'standard' : null;
};

/** 生成要求 → 一段给模型看的话。每一个生成环节都拼在自己的提示词里 */
export function renderSpec(spec?: GenSpec | null, anchor?: Anchor | null): string {
  if (!hasSpec(spec) && !anchor?.from) return '';
  const s = spec || {};
  const lines: string[] = ['【用户的生成要求（每一步都要遵守；和上面的固定口径冲突时，以这里为准）】'];
  if (anchor?.from && anchor.from !== anchor.label) lines.push(`· 用户最初说的是「${anchor.from}」，已经定位到「${anchor.label}」这一级：就按「${anchor.label}」来做，不要回到更细、更资深的岗位上。`);
  if (s.form && FORMS[s.form]) { const f = FORMS[s.form]; lines.push(`· 形态：${f.label}——${f.essence}。体验者是${f.who}。工位：${f.bench}评分：${f.score}`); }
  if (s.audience) lines.push(`· 给谁看：${s.audience}`);
  if (s.depth) lines.push(`· 深度：${s.depth}`);
  if (s.difficulty) lines.push(`· 考核难度：${s.difficulty}`);
  if (s.must) lines.push(`· 必须放进去：${s.must}`);
  if (s.avoid) lines.push(`· 不要：${s.avoid}`);
  if (s.refs?.length) {
    let left = REF_BUDGET;
    const parts: string[] = [];
    for (const r of s.refs) { if (left <= 200) break; const t = r.text.slice(0, Math.min(left, 2500)); left -= t.length; parts.push(`《${r.name}》\n${t}`); }
    lines.push(`· 参考资料（照着它的场景、流程、说法来，不要照抄成题目原文）：\n${parts.join('\n---\n')}`);
  }
  return lines.join('\n');
}

/** 用户 / 前端传来的要求做清洗：只留认识的键、截长度 */
export function cleanSpec(raw: any): GenSpec {
  const t = (v: any, n: number) => (typeof v === 'string' ? v.trim().slice(0, n) : '');
  const refs = Array.isArray(raw?.refs) ? raw.refs.slice(0, 6).map((r: any) => ({ name: t(r?.name, 80) || '参考资料', text: t(r?.text, 12000) })).filter((r: SpecRef) => r.text.length > 20) : [];
  const out: GenSpec = { preset: t(raw?.preset, 40) || undefined, form: isForm(raw?.form) ? raw.form : undefined, audience: t(raw?.audience, 600), depth: t(raw?.depth, 800), difficulty: t(raw?.difficulty, 800), must: t(raw?.must, 1500), avoid: t(raw?.avoid, 800), refs, updated_at: new Date().toISOString() };
  for (const k of Object.keys(out) as (keyof GenSpec)[]) if (out[k] === undefined || out[k] === '' || (Array.isArray(out[k]) && !(out[k] as any[]).length)) delete out[k];
  return out;
}

    p.role ? `数字职人（主角的师傅 / 化身）：${p.name || ''} · ${p.role}${p.tagline ? `，口头禅「${p.tagline}」` : ''}` : '',
    jd.responsibilities ? `岗位职责：${String(jd.responsibilities).slice(0, 800)}` : '',
    career.summary ? `职业概况：${String(career.summary).slice(0, 400)}` : '',
    specTextOf(jd),   // 用户的生成要求（受众 / 深度 / 难度 / 必须有 / 不要 / 参考资料），每一步都带
  ].filter(Boolean).join('\n');
}

/**
 * 生涯体验讲「怎么进这一行」用职业库里的事实（月薪、相关专业、证书、发展阶梯），别让模型凭空编（11-06）。
 * 只给 career 形态；空间没挂职业库就空串。
 */
async function occupationFacts(space: any): Promise<string> {
  const id = space?.jd_snapshot?.anchor?.occupation_id;
  if (!id || formOf(space?.jd_snapshot) !== 'career') return '';
  const { data: o } = await supabaseAdmin.from('occupations').select('name, recent_year_monthly_salary, relevant_certificates, relevant_fos_ids, position_requirement, career_map').eq('id', id).maybeSingle();
  if (!o) return '';
  const ladder = (o.career_map as any)?.ladder;
  return ['【职业库里的事实（讲收入、专业、证书、发展时先用这些；但只取和这一行直接对口的——库里的专业 / 证书明显不对口就不写，换成常见的对口专业和这一行的职业技能等级证书，并注明「以各校招生章程 / 实际为准」；不要把不对口的条目原样念给学生）】',
    o.recent_year_monthly_salary ? `· 近一年平均月薪约 ${o.recent_year_monthly_salary} 元（全国口径，各地差别大）` : '',
    o.relevant_fos_ids?.length ? `· 相关专业（学科代码，写给学生时换成中文专业名）：${o.relevant_fos_ids.join('、')}` : '',
    o.relevant_certificates ? `· 相关证书：${o.relevant_certificates}` : '',
    o.position_requirement ? `· 招聘常见要求：${String(o.position_requirement).slice(0, 240)}` : '',
    Array.isArray(ladder) && ladder.length ? `· 发展阶梯：${ladder.map((x: any) => x?.title || x?.name || x).filter(Boolean).join(' → ').slice(0, 300)}` : '',
  ].filter(Boolean).join('\n');
}

const castText = (cast: CastMember[]) => (['person', 'place', 'prop'] as CastKind[]).map(k => {
  const list = cast.filter(m => m.kind === k && !m.is_self);
  return `${castKind(k).label}：${list.length ? list.map(m => `${m.code} ${m.name}（${m.type_name || '未定类型'}${m.note ? `；${m.note}` : ''}）`).join('；') : '（还没有）'}`;
}).join('\n');

const chapterLine = (c: StudioChapter) =>

const str = (v: any, n: number) => String(v ?? '').trim().slice(0, n);
const KINDS = ['daily', 'incident', 'assessment'];

/** 新角色进角色表：先按规范类型名找素材库能复用的；道具找不到就建一张道具卡归库（先不画图） */
async function addCast(taskId: string, cast: CastMember[], x: { kind: CastKind; name: string; type_name: string; note: string; look: string }, family: string): Promise<CastMember> {
  const same = cast.find(m => m.kind === x.kind && m.name === x.name);
  if (same) return same;
  let asset_id: string | null = null;
  const hit = await findReusable(castKind(x.kind).asset, x.type_name, family, cast.map(m => m.asset_id));
  if (hit) { asset_id = hit.id; await supabaseAdmin.from('lab_art_assets').update({ uses: (hit.uses || 1) + 1 }).eq('id', hit.id); }
  else if (x.kind === 'prop') asset_id = (await registerAsset({ kind: 'prop', title: x.name, type_name: x.type_name || x.name, note: x.look, family, source_task_id: taskId })).id;
  const { data, error } = await supabaseAdmin.from('lab_cast').insert({
    task_id: taskId, code: nextCastCode(cast, x.kind), kind: x.kind, name: x.name, type_name: x.type_name, note: x.note, look: x.look, asset_id,
  }).select('*, asset:lab_art_assets(*)').single();
  if (error) throw error;
  cast.push(data as CastMember);
  return data as CastMember;
}

/** 「师傅 & 你」「你和老周」这种把体验者并进去的名字，只留对方（11-13 反馈：立绘画成两个人、台词说话人写成「师傅 & 你」） */
const noYou = (s: string) => String(s || '').replace(/\s*[&＆和与]\s*你$/, '').replace(/^你\s*[&＆和与]\s*/, '').trim();

function readNewCast(raw: any): { tmp: string; kind: CastKind; name: string; type_name: string; note: string; look: string }[] {
  return (Array.isArray(raw) ? raw : []).map((c: any, i: number) => ({
    tmp: str(c.id || c.tmp || `N${i + 1}`, 10), kind: CAST_K.includes(c.kind) ? c.kind : 'person',
    // 模型偶尔把编号填进名字（name: "T01"）：这种用类型名当名字
    name: noYou(/^[PSTN]\d{1,2}$/.test(str(c.name, 30)) ? str(c.type_name, 30) : str(c.name, 30)), type_name: noYou(str(c.type_name, 30)), note: str(c.note, 120), look: str(c.look, 300),
  })).filter(c => c.name).slice(0, 8);
}

async function buildDayInner(spaceId: string, model: string, hint: string, progress: Progress, plan: { slot: string; text: string }[]): Promise<{ added: number; total: number; cast: number }> {
  const { data: space, error } = await supabaseAdmin.from('skill_tasks').select('id, title, profile, jd_snapshot, bible').eq('id', spaceId).single();
  if (error) throw error;
  // 重排骨架：空的、没锁的草稿格子先清掉（有步骤的、锁定的、已发布的都留着）

${spaceText(space)}
${facts}

${chapters.length ? chapters.map(chapterLine).join('\n') : '（还没有）'}

${castText(cast)}

${DAY_RULES}
${hint ? `\n管理员的要求：${hint}\n` : ''}

${plan.length ? `- 这个空间的形态是「${form ? FORMS[form].label : '新人上班的一天'}」${form ? `（${FORMS[form].essence}），体验者是${FORMS[form].who}` : ''}。

- 这一天会反复出场、但角色表里还没有的人物 / 地点 / 道具，放进 new_cast（各章生成时会优先用）。人物 2–6 个，地点 2–4 个，道具 2–5 个（道具一定要有）。
${CAST_RULES}

- 生涯体验要有一条主线（11-20 照百业同事的样板）：facts 第一条写「今天的主线事件」——一件贯穿全天的事（如「昨夜急诊收的阑尾炎患者小许，今天第一台手术就是他」「上午线上白屏，祸源是昨天的发版」）；第二条写「全天最大的扣子」——上午的哪个判断 / 教训，到哪一段被验证或成为做决定的理由。人物定一位带教师傅（有名字、有性格）+ 2–3 位协作的人。` : ''}${form === 'training' || form === 'assess' || form === 'certify' ? `

- 全部用中文。

返回 JSON：
{
  "existing": [ { "id": "<已有章节 id>", "slot": "<给它的时段>" } ],
  "slots": [ { "slot": "08:30", "title": "<格子标题，10 字左右>", "kind": "daily|incident|assessment", "brief": "<这一章要交代的事，1–2 句，写清和前后章节的衔接>" } ],
  "new_cast": [ { "kind": "person|place|prop", "name": "", "type_name": "", "note": "", "look": "" } ],

}`, model, 'Lab Day Skeleton');

  // 已有章节放进时段（没写时段的才填，人写过的不覆盖）
  const ids = new Set(chapters.map(c => c.id));
  for (const e of Array.isArray(p.existing) ? p.existing : []) {
    const c = chapters.find(x => x.id === e?.id);
    if (c && ids.has(c.id) && !c.slot && str(e.slot, 40)) { c.slot = str(e.slot, 40); await supabaseAdmin.from('lab_chapters').update({ slot: c.slot }).eq('id', c.id); }
    // 人排了时间点（11-19 实测：建空间自带的那章被放进 13:30，标题还是「前端组件设计工作台」）：没锁的草稿章改用人写的标题和要交代的事，之后重写这一章就按它来
    const pl = c && plan.find(x => x.slot === c.slot);
    if (c && pl && !c.locked && c.status === 'draft') { const [t0, ...rest] = pl.text.split('｜'); const title = (rest.length ? t0 : pl.text).slice(0, rest.length ? 20 : 8); const brief = rest.length ? rest.join('，') : pl.text; c.title = title; (c as any).brief = brief; await supabaseAdmin.from('lab_chapters').update({ title, brief }).eq('id', c.id); }
  }
  if (form === 'certify') { const daily = chapters.filter(c => c.kind !== 'assessment' && !c.locked).map(c => c.id); if (daily.length) await supabaseAdmin.from('lab_chapters').update({ kind: 'assessment' }).in('id', daily); }
  // 新角色先进角色表，好把简介里的编号（P01 / 临时的 N1）换成名字——简介是给体验者看的
  await progress('登记角色', '新的人 / 地方 / 道具先进角色表，同类型的从素材库复用');
  const before = cast.length;
  const tmpDay = new Map<string, CastMember>();
  for (const x of readNewCast(p.new_cast)) tmpDay.set(x.tmp, await addCast(spaceId, cast, x, family));
  const nameOf = (code: string) => (tmpDay.get(code) || cast.find(m => m.code === code))?.name;

    // 资质考核每一场都是考核（11-06 国内版：电工「考核3」标成了日常）
    .map((x: any) => ({ slot: str(x.slot, 40), title: decodeCastRefs(str(x.title, 60), nameOf), kind: form === 'certify' ? 'assessment' : KINDS.includes(x.kind) ? x.kind : 'daily', brief: decodeCastRefs(str(x.brief, 400), nameOf) }))
    .filter((x: any) => x.slot && x.title)
    // 已有章节占了的时段不再新建（11-06 考核样例：建空间时自带的工位章排到「考核4」，又新建了一格「考核4」做同一件事）
    .filter((x: any) => !chapters.some(c => (c.slot || (Array.isArray(p.existing) ? p.existing.find((e: any) => e?.id === c.id)?.slot : '')) === x.slot))
    // 生涯体验一天 5–6 格（含已有章节）（11-13 反馈：说排了 5 格，实际出来 6 章，建空间自带的那章还是「未定时段」）

  // 人排好的格子（11-19）：时段和格数以人为准——模型漏的补上（标题取那句话的开头），多出来的去掉；已有章节占了的时段不再建
  if (plan.length) {
    const taken = new Set(chapters.map(c => c.slot).filter(Boolean));
    slots = plan.filter(x => !taken.has(x.slot)).map(x => {
      const m = slots.find((y: any) => y.slot === x.slot);
      const [t0, ...rest] = x.text.split('｜');
      return m ? { ...m, slot: x.slot } : { slot: x.slot, title: (rest.length ? t0 : x.text).slice(0, 8), kind: form === 'certify' ? 'assessment' : 'daily', brief: rest.length ? rest.join('，') : x.text };
    });
  }
  // 生涯体验最后一格必须是「我和这一行」（讲选科、对口专业、证书、怎么入行）：模型漏了就补上
  if (form === 'career' && !plan.length && !(want && want <= 4) && !slots.some((x: any) => x.kind === 'assessment' || /我和这一行/.test(x.title)) && !chapters.some(c => c.kind === 'assessment' || /我和这一行/.test(c.title))) {
    if (slots.length && slots.length + chapters.length >= (want || 6)) slots.pop();
    slots.push({ slot: '17:30', title: '收尾：我和这一行', kind: 'assessment', brief: '回看这一天，聊聊自己合不合适；讲清高中怎么选科、对口的大学 / 高职专业、要考的证书和入行的路。' });
  }
  // 已有章节还没有时段的（模型没给），放进新格子中间空着的时段，别留「未定时段」
  const used = [...chapters.map(c => c.slot).filter(Boolean), ...slots.map((x: any) => x.slot)].map(x => slotMinutes(x as string)).filter((x): x is number => x != null).sort((a, b) => a - b);
  for (const c of chapters.filter(c => !c.slot)) {
    let best = 600, gap = -1;   // 默认 10:00
    for (let i = 0; i + 1 < used.length; i++) if (used[i + 1] - used[i] > gap) { gap = used[i + 1] - used[i]; best = Math.round((used[i] + used[i + 1]) / 2 / 15) * 15; }
    c.slot = `${String(Math.floor(best / 60)).padStart(2, '0')}:${String(best % 60).padStart(2, '0')}`; used.push(best); used.sort((a, b) => a - b);
    await supabaseAdmin.from('lab_chapters').update({ slot: c.slot }).eq('id', c.id);
  }
  let seq = Math.max(0, ...chapters.map(c => c.seq));
  if (slots.length) {
    const { error: e2 } = await supabaseAdmin.from('lab_chapters').insert(slots.map((x: any) => ({
      task_id: spaceId, seq: ++seq, slot: x.slot, title: x.title, kind: x.kind, brief: x.brief, sim: { title: x.title, intro: '', steps: [] }, status: 'draft',
    })));
    if (e2) throw e2;
  }
  const bible: DayBible = {
    ...((space as any).bible || {}),
    facts: (Array.isArray(p.facts) ? p.facts : []).map((f: any) => decodeCastRefs(str(f, 300), nameOf)).filter(Boolean).slice(0, 8),
    outline: [...chapters.map(c => ({ slot: c.slot || '', title: c.title })), ...slots.map((x: any) => ({ slot: x.slot, title: x.title }))]
      .sort((a, b) => (slotMinutes(a.slot) ?? 9999) - (slotMinutes(b.slot) ?? 9999)),
    updated_at: new Date().toISOString(),
  };
  await supabaseAdmin.from('skill_tasks').update({ bible }).eq('id', spaceId);
  return { added: slots.length, total: chapters.length + slots.length, cast: cast.length - before };
}

/** 逐格生成一章（记账：记到这个空间、这一章上，含补画的人物 / 场景） */
export function buildChapter(spaceId: string, chapterId: string, model: string, hint: string, opts: { art?: boolean }, progress: Progress): Promise<{ steps: number; newCast: number; drawn: number; reused: number }> {
  return withCost({ space_id: spaceId, chapter_id: chapterId, stage: 'create' }, () => buildChapterInner(spaceId, chapterId, model, hint, opts, progress));
}
async function buildChapterInner(spaceId: string, chapterId: string, model: string, hint: string, opts: { art?: boolean }, progress: Progress): Promise<{ steps: number; newCast: number; drawn: number; reused: number }> {
  const [{ data: space, error }, { data: chs, error: e2 }] = await Promise.all([
    supabaseAdmin.from('skill_tasks').select('id, title, profile, jd_snapshot, bible').eq('id', spaceId).single(),
    supabaseAdmin.from('lab_chapters').select('*').eq('task_id', spaceId).order('seq'),
  ]);
  if (error) throw error;
  if (e2) throw e2;
  const ch = (chs || []).find((c: any) => c.id === chapterId) as StudioChapter | undefined;
  if (!ch) throw new Error('章节不存在');
  if (ch.locked) throw new Error('这一章锁定了：先解锁再重新生成');
  const cast = await loadCast(spaceId);
  const family = familyOf((space as any).jd_snapshot?.career?.profession, (space as any).jd_snapshot?.title);
  const form = formOf((space as any).jd_snapshot);
  const facts = await occupationFacts(space);
  const bible: DayBible = (space as any).bible || {};

/** 不用 sanitizeSim：它会悄悄丢步骤、截到 8 步。这里补齐结构、把角色编号换成角色表里的人名 / id，问题留给工作室的校验去标 */
function cleanSim(raw: any, ch: StudioChapter, ref: (v: any) => CastMember | null, careerEnd?: 'none' | 'reflect', role = ''): Sim {
  const TYPES = ['choose', 'multi', 'classify', 'allocate', 'slider', 'text'];
  const used = new Set<string>();

    while (used.has(id)) id += 'x';
    used.add(id);
    const st: SimStep = { id, type: s.type, prompt: String(s.prompt) };
    if (s.scene?.text) {
      const who = ref(s.scene.who);
      st.scene = { who: oneWho(who?.kind === 'person' ? who.name : s.scene.who) || '同事', time: s.scene.time ? String(s.scene.time) : undefined, text: String(s.scene.text) };
    }
    const place = ref(s.place);
    if (place?.kind === 'place') st.place = place.id;
    const props = (Array.isArray(s.props) ? s.props : []).map(ref).filter((m: CastMember | null) => m?.kind === 'prop').map((m: CastMember | null) => m!.id);
    if (props.length) st.props = [...new Set(props)] as string[];

    if (s.type === 'allocate') { st.total = Math.max(1, Number(s.total) || 100); st.unit = String(s.unit || ''); }
    if (s.type === 'slider') { st.min = 0; st.maxValue = 100; st.unit = String(s.unit || '%'); }

/** 评分标准：4–5 个维度，权重归一到 100；不合格就返回 null（评分时用通用四项） */
export function normalizeRubric(raw: any): { key: string; name: string; weight: number; description: string }[] | null {
  let list = (Array.isArray(raw) ? raw : []).map((r: any, i: number) => ({
    key: String(r?.key || `d${i + 1}`).replace(/[^\w-]/g, '').slice(0, 24) || `d${i + 1}`, name: str(r?.name, 20), weight: Math.max(0, Math.round(Number(r?.weight) || 0)), description: str(r?.description, 200),
  })).filter(r => r.name && r.weight > 0).slice(0, 5);
  if (list.length < 3) return null;
  const total = list.reduce((a, r) => a + r.weight, 0);
  list = list.map(r => ({ ...r, weight: Math.round((r.weight / total) * 100) }));
  list[0].weight += 100 - list.reduce((a, r) => a + r.weight, 0);
  return list;
}

**反馈**：D 最有诱惑力——条文确实要查，但师傅问的是「裁判口径」：法院认不认、赔多少，判决书才回答这个问题。热帖和新闻只能当背景，不能当依据。

**Q3｜模块：接案判断（多选）**
林女士被请回来，师傅和她谈了十五分钟才签委托。小陈悄声问你：师傅接案前在确认什么？

- A. 这个案子大概能不能赢
- B. 客户的诉求现实吗
- C. 收费够不够覆盖投入的时间
- D. 案子发到网上会不会火

**正确答案**：ABC。
**反馈**：D 是最常见的错觉——律师按专业判断和档期选案，不按流量。接案本身就是一道专业题：算胜算、算成本、算时间。

---

**【段末】**
委托签下。师傅按你的三道作答生成点评（同原版「写好了，看看师傅怎么说」机制），然后说：
> 「明天她把材料袋送来，真正的活儿才开始。」

## 第二段 · 11:20 材料整理（约15分钟）

**【开场剧情】**
材料袋上桌：劳动合同、工资条、微信打印件、录音转写稿。师傅让你建证据目录——把每份纸变成能用的东西。

---

**Q4｜模块：证据整理**
哪一项「材料 → 能证明什么」的配对是对的？

- A. 劳动合同 → 她和公司存在劳动关系 ✅
- B. 录音「你明天不用来了」→ 公司违法辞退了她
- C. 微信里王经理回的「收到」→ 公司批准了病假
- D. 工资条 → 她是被口头辞退的

**反馈**：B 越界——录音只证明「王经理说过这话」，还不等于「这是公司的决定」，每份材料只能写它真撑得住的那一句；C 是「收到 ≠ 批准」；D 答非所问，工资条证明的是劳动关系和月薪标准。**本段因果核心**：正因为发现录音还差一环，才有下一题的决策。

---

**Q5｜模块：文书写作**
录音的缺口补不上，师傅决定先发律师函、给公司七日期限，同时把协商的门留着。函的结尾句你来定：

- A. 「请贵司于收函七日内书面答复，否则我方将依法申请劳动仲裁。」✅
- B. 「请贵司慎重考虑，尽快处理为盼。」
- C. 「若七日内不答复，我方将采取一切手段维权！」
- D. 「你们等着吃官司吧。」

**反馈**：B 没有期限，等于没说；C、D 的狠话不留余地，还可能被对方抓住把柄。A 的每个词都有功能：期限、书面、为下一步程序铺垫。律师的输出物是文件，文件里没有一句是白写的。15:00 函件发出。

---

**Q6｜模块：电话沟通**
函发出后，16:00 你要给对方 HR 打电话跟进。师傅让你先写通话预案。哪一份合格？

- A. 「目标：争取十日内答复｜开场：报身份和来意｜预判：她说『在走流程』→ 追问具体期限」✅
- B. 只写一行：「下午四点打给 HR」
- C. 写了一整页逐字稿，打算背下来照念
- D. 「目标：把她说服，让她认错」

**反馈**：B 只是日程不是预案；C 一被岔开就乱；D 目标本身不现实——律师的电话是带预案的沟通，不靠临场发挥也不靠说服。电话打完，HR 答应「本周五前给方案」。

## 第三段 · 15:30 案件研判（约18分钟）

**【开场剧情】**
白板前，师傅把你和小陈叫来：「各说各的方案，再合议。」

---

**Q7｜模块：团队研判**
三个方案，哪个风险最低？

- A. 先约公司协商，同时备齐仲裁材料，谈不拢再仲裁 ✅
- B. 不联系公司，直接申请仲裁，一步到位
- C. 先等公司主动认错，再谈赔偿

**反馈**：B 舍弃了协商这条更快、更体面的路；C 把主动权全交给对方。A 进可攻退可守——研判的价值就在把每条路的代价摆出来再选。合议定了 A，HR 约了面谈。

---

**Q8｜模块：收尾分寸**
方案定了，你以为可以下班。师傅拉出一张当日收尾清单——哪件事今天**不该**做？

- A. 给林女士发「下一步时间表」
- B. 把今天的材料整理归档
- C. 立刻向仲裁委递交申请 ✅
- D. 开具本案的收费票据

**反馈**：方案是「先协商」，递交申请是协商破裂后的牌——今天的活是把它**准备好**，不是把它做掉。律师的一天由大量这种「分寸」组成。

---

**Q9｜模块：工作归类（收口题）**
19:30，材料归档完毕。锁门前师傅问你最后一个问题：哪一组活全属于律师的「对内」工作？

- A. 接待咨询、谈案收费、汇报进展
- B. 证据整理、团队研判、归档开票 ✅
- C. 法律检索、发律师函、电话跟进
- D. 出庭、谈判、取证

**反馈**：A 是对客户、C 是对外。答完这道收口题，今天的 9 个模块在心里过完最后一遍——这一天的班，值完了。

---

**【段末收尾场景 · 19:40 下班路上 · 全天唯一的反思题】**

答完 Q9，师傅没有立刻说再见，递给你一杯水：
> 「今天最后一步不是答题。跑完这一天，写一句你的反思——一天的感受，写一次才真实。」

**反思题**（自由作答，无对错，限 80 字）：

> **「走完律师的一天：哪件事你干得最有劲？哪件事你撑不住？这样的每一天，你还愿意吗？」**

题面里三小问只是提示方向，**合并成一个输入框、一次写完**——不拆成三道，避免段段打卡式的疲劳；「不用讨好任何人，实话就行」的提示放在输入框 placeholder 里。

提交后，师傅生成一段回应（AI 根据反思内容生成，呼应学生自己的用词，不套模板）：
> 「你说检索最有劲——这说明你喜欢『用证据说话』的感觉，这是这行的底色；但你说电话预案难熬——许多律师刚入行都这样，预案写多了就熟了。你最后那句『愿意再试一次』，比答对九道题更让我高兴。」

- **「师傅寄语」**：AI 依据反思内容生成的个性化评语

学生带走的不是一张「通过证书」，而是一份「今天的我」的记录。

- 反思题仅一道，全天结束后写一次——一天的感受写一次才真实；自由作答、不做成选择题，感受一旦做成选项就变回「测性格」，写出来才是自己的

# 《外科医生的一天》职业体验·完整题目设计

> 面向高中生的职业模拟（AI 百业 · 数字职人 NOVA-94 带教），与《律师的一天》《前端工程师的一天》同一套结构。
> 全天 3 段、每段 3 题，共 9 道决策题 + 1 道收尾反思（全天结束时写，仅此一次）。
> 设计原则：① 每题对应外科日常的一个真实工作模块；② 选项有像样的干扰项，答错有反馈；③ 题目之间因果串联，答案推动剧情；④ 全天只写一次反思，放在结束处。
> 说明：剧情为虚构教学场景，只为呈现职业日常，不构成任何医疗建议；不出现血腥描写。

**主线人物**：老陆（带教导师、主治医师）、「你」（体验者）、小夏（住院医师）、马医生（麻醉医师）、蔡姐（器械护士/护理组长）。
**今天的主线事件**：昨夜急诊收了一位急性阑尾炎患者小许（22 岁、大学生），今天第一台手术就是他——上午上台，下午管回病房，晚上你还要跟他一起去会诊。

## 第一段 · 07:30 交班与术前准备（约15分钟）

**【开场剧情】**
07:30 医生办公室，晨间交班。夜班小夏交接：22 岁患者小许，转移性右下腹痛 8 小时，血象偏高。老陆把病历夹推给你：「今天第一台是他。上台之前，功课先做足。」

---

**Q1｜模块：术前评估（读片）**
上台前，你的「功课」是什么？

- A. 问小许一句「还疼吗」，疼就做，不疼就等等
- B. 把血检、B 超和 CT 过一遍，确认诊断成立、有没有手术禁忌，评估穿孔风险 ✅
- C. 信任急诊的诊断，直接上台

**反馈**：C 是把别人的判断当成自己的责任——上了台，签字的是主刀；A 是拿症状代替证据。B 是外科的规矩：**手在后，眼和脑在前**。你读片时注意到粪石和局部渗出，老陆点头：「穿孔风险不低，别拖，排今天第一台。」——这个判断晚上还会被验证一次。

---

**Q2｜模块：术前谈话**
小许妈妈攥着单子问：「医生，这手术一定安全吧？」老陆让你先答：

- A. 「小手术，放心，肯定没问题。」
- B. 如实讲：手术怎么做、预期获益、可能的风险（感染、出血、麻醉意外）和替代方案，讲完签知情同意书 ✅
- C. 「这些让护士跟您解释，我还要去看台子。」

**反馈**：A 是给纠纷埋雷——「没问题」三个字，出任何一点异常都会变成「你们说过没问题」；C 是把本职推出去，谈话是外科医生的法定动作。B 的坦白不会吓跑家属，反而换来配合：**把最坏的可能讲在前面，才配得上家属的托付。** 记住小许妈妈签字时问的那句「术后多久能正常吃饭」——傍晚查房它会再来一次。

---

**Q3｜模块：手术安全核查**
10:00 手术室。消毒铺巾之前，蔡姐按核查表开始逐项核对。这一步是干什么的？

- A. 三方（手术医生、麻醉、护理）核对患者身份、手术部位、禁食情况、过敏史，逐项口头确认 ✅
- B. 走个过场，反正病历上写着呢
- C. 让患者自己报姓名和手术部位就够了

**反馈**：B 是拿别人的教训不当回事——安全核查表里的每一项，背后都是真实的事故：开错部位、没禁食导致麻醉呕吐误吸、药物过敏。**它看起来琐碎，却是台上所有流程里最不能省的一关。** 核查通过，无影灯亮起。

## 第二段 · 10:30 手术台上（约15分钟）

**【开场剧情】**
你站在第二助手位。老陆主刀，马医生盯麻醉，蔡姐递器械。你负责拉钩暴露术野——听起来最简单的活。

---

**Q4｜模块：无菌原则**
摆体位时，你的手肘不小心蹭到了铺巾边缘的非无菌区。没人注意到。你——

- A. 不动声色继续，免得耽误手术、显得外行
- B. 立刻出声报告，退后重新洗手、更换手套 ✅
- C. 心里记着那块地方，之后避开它操作就行

**反馈**：A 和 C 是同一种侥幸——「可能没沾上」在无菌原则里等于「沾上了」；被污染的器械和术野是看不见的，等到表现为术后感染，追查时谁也说不清。台上没人会嘲笑重新洗手的人，**隐瞒才会**。你重新消毒用了三分钟，手术延时三分钟——这三分钟是这个职业里最便宜的保险。

---

**Q5｜模块：助手职责**
老陆叮嘱你：「钩子拉稳，眼睛别闲着。」助手在台上真正的价值是什么？

- A. 只管拉钩，其他是主刀和麻醉的事
- B. 牵拉保持稳定，同时留意监护数值、术野出血和冲洗液颜色，有变化马上出声 ✅
- C. 频繁凑近看，随时向主刀提问

**反馈**：A 是把自己当工具，C 是把自己当观众——B 才是助手：**稳，是给主刀一个不动的术野；看，是给整个团队多一双眼睛**。手术是流水线上的团队作战，每个人的专注都是别人的安全冗余。

---

**Q6｜模块：术中应对**
19 分钟时，马医生突然出声：「心率 118，血压有点往下走。」台上正确的反应是？

- A. 各盯各的仪表盘，异常必须马上说出来，主刀统一决策，该暂停就暂停 ✅
- B. 小声商量，别惊动主刀
- C. 麻醉的事归麻醉，外科继续做外科的

**反馈**：B 是把隐患埋进沉默里；C 拆散了团队。A 是台上的铁律：**信息必须全team共享，决策只有一个出口**。老陆暂停探查——果然如术前读片判断的：阑尾已早期穿孔，腹腔少量脓性渗出。冲洗、引流、术后抗感染加强，术中所见印证了你上午的评估。这个「验证」是外科一天里最踏实的时刻之一。

## 第三段 · 15:30 术后与值班（约18分钟）

**【开场剧情】**
13:00 手术顺利结束，你以为能松口气了。老陆把你叫回办公室：「手术刀只占外科医生一天的一小半。走，还有一堆活。」

**Q7｜模块：术后医嘱与病历**
回到病房，第一件事是什么？

- A. 开术后医嘱（禁食、补液、抗感染、引流管护理），补手术记录和病程记录 ✅
- B. 交代护士一句「手术很顺利」，先去吃个饭
- C. 晚上值完班再补，反正都记得

**反馈**：C 是新医生的通病——病历不是「记录」，它是**下一班医护的信息接力棒**，也是法律文件；漏掉一条过敏史，夜班的同事就可能踩坑。A 看起来是文书活，实际上是「手术的成果能不能安全落地」的另一半。

---

**Q8｜模块：术后观察**
17:30 查房。小许说：「伤口有点疼，还有点心慌。」你——

- A. 「术后都这样，忍忍就好。」
- B. 测生命体征、看引流液颜色和量和腹体征，对照术后正常恢复过程逐项排查，有疑问立即报告老陆 ✅
- C. 直接开一支止痛药，别让家属担心

**反馈**：A 和 C 都是把「看起来正常的疼」放过去了——术后最危险的并发症（出血、感染）恰恰都藏在术后疼痛和心率变化里。B 是外科的「照妖镜」流程：**每一项主诉都要回到客观数据上过一遍**。小许指标平稳，是正常术后反应——但这次排查本身不能省。上午 Q2 谈话时讲过的「可能的风险」，此刻就是你判断的清单。

**Q9｜模块：值班会诊与归类（收口题）**

- A. 晨间交班、急诊会诊、与麻醉和护理核对安全核查表 ✅
- B. 查房、开医嘱、术前谈话（这是对患者）
- C. 写手术记录、整理病历（这是对内文书）
- D. 主刀操作、拉钩暴露、缝合（这是对患者）

**反馈**：外科医生的一天由三张网组成：对患者（查房、谈话、手术）、对协作（交班、会诊、核查）、对内（病历、读片、复盘）。答完这道收口题，今天的 9 个模块在心里过完最后一遍——这一天的班，值完了。

# 《前端工程师的一天》职业体验·完整题目设计

> 面向高中生的职业模拟（AI 百业 · 数字职人 NOVA-94 带教），与《律师的一天》同一套结构。
> 全天 3 段、每段 3 题，共 9 道决策题 + 1 道收尾反思（全天结束时写，仅此一次）。
> 设计原则：① 每题对应前端日常的一个真实工作模块；② 选项有像样的干扰项，答错有反馈；③ 题目之间因果串联，答案推动剧情；④ 全天只写一次反思，放在结束处。

**主线人物**：Ada 姐（带教导师、资深前端）、「你」（体验者）、王姐（产品经理）、大刘（后端工程师）、小周（测试）。
**今天的主线事件**：上午一个线上白屏事故——祸源恰恰是昨天的发版；下午的新页面，晚上要和修复一起面对「发不发、怎么发」的决定。

## 第一段 · 09:30 线上救火（约15分钟）

**【开场剧情】**
09:00 你到工位，站会上 Ada 姐把今天的任务排给你。话音刚落，客服群炸了：不少用户反馈商城页打开全白。Ada 姐看了一眼监控曲线，把椅子滑到你旁边：「先别慌，跟着我一步步来。」

**Q1｜模块：复现问题**
用户只留了一句「页面白屏」。动手改代码之前，你的第一步是？

- A. 凭经验猜是哪里坏了，直接改
- B. 先复现：用用户同款的手机和浏览器，打开同一个页面看 ✅
- C. 回复用户「麻烦重启手机再试试」

**反馈**：A 是新手最常见的坑——没复现就改，改 ten 次也不一定中；C 能解决个例，但问题还在线上。**稳定复现是修 bug 的第一步**：能造出来，才能找原因。

---

**Q2｜模块：定位原因**
复现成功，控制台报了一条错。怎么把「报错」变成「原因」？

- A. 把报错截图发群里，等大佬认领
- B. 顺着报错堆栈找到出错的那行代码，再对比昨天的发版改了什么 ✅
- C. 重装浏览器，说不定就好了

**反馈**：C 只对环境问题有效；A 是把定位责任推出去。B 是标准动作：**堆栈告诉你错在哪行，「对比昨天的改动」告诉你错为什么今天才出现**——你顺着查下去，发现昨晚上线的活动页用了新语法，老版本浏览器不认识。

---

**Q3｜模块：修复决策**
原因确认：昨天的活动页用新语法赶了上线，老浏览器直接崩。怎么修？

- A. 改成兼容写法，小周提测通过后发版 ✅
- B. 在页面上提示「请升级浏览器」
- C. 把昨晚的发版整个回滚

**反馈**：B 是把修复责任推给用户——那部分用户还是丢了；C 能止血但太重，活动页也一起没了。A 是老练的做法：**修最小范围、走完提测、再发版**。注意——正是因为今天栽在「发版」上，晚上你还要面对一次发版决策（第三段）。

**Q4｜模块：还原设计稿**
设计稿上标得清清楚楚：按钮圆角 8px、间距 16px、主色 #4F46E5。你写到什么程度算「做完了」？

- A. 看着差不多就行，像素级的偏差用户看不出来
- B. 按标注逐项还原，再用工具和设计稿叠加对比，调到重合 ✅
- C. 顺手加几个设计稿上没有的动画，显得更精致

**反馈**：A 的「差不多」会在设计师验收时一条条打回来；C 的好心发挥破坏的是协作规则——**设计是别人的专业，还原是前端的本分**。B 看起来笨，却是这行说的「像素眼」基本功。

**Q5｜模块：接口联调**
页面写完了，一调大刘的接口——约定好的 `price` 字段没返回，页面价格全空。你——

- A. 自己在代码里写死一个价格顶上去，先跑起来再说
- B. 找大刘对齐：是接口漏了字段，还是约定变了？确认改哪边 ✅
- C. 直接登录后端代码库，帮他改掉

**反馈**：A 最隐蔽也最危险——**悄悄补数据等于把问题埋进代码里**，以后接口真出错也发现不了；C 越权了，后端不是你能随便改的；B 才是协作：接口约定是两个人的合同，谁改都要摆在明面上。对完发现是版本合并时字段丢了，大刘十分钟修好。

**Q6｜模块：自测检查（多选）**

- B. 模拟接口报错和断网，看页面会不会白屏
- C. 直接发到线上，让用户帮忙测

**正确答案**：AB。
**反馈**：C 是拿线上用户当测试员——上午的白屏就是这么来的；**自己测过、异常场景也测过**，才有资格谈发版。这一步做完，Ada 姐在提测单上签了字。
